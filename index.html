Drop into src/layers/sargassum/index.js to add the Veabird + CONABIO Marine Layer: 
/**
 * Cancún & Mexican Caribbean Sargassum + Marine Protected Areas Layer
 * Sources:
 *  - Veabird Caribbean Sargassum Satellite Record (https://veabird.com/datasets/sargassum/sargassum-monthly.json)
 *  - Quintana Roo National Marine Parks API (https://poidata.io/api/national-park/quintana-roo/mexico)
 */
export function createSargassumLayer({ viewer, Cesium }) {
  let enabled = false;
  const entities = [];

  async function update() {
    if (!enabled) return;
    const res = await fetch('/api/cancun/sargassum');
    const data = await res.json();
    entities.forEach((e) => viewer.entities.remove(e));
    entities.length = 0;

    for (const seg of data.segments) {
      entities.push(
        viewer.entities.add({
          name: seg.name,
          polyline: {
            positions: Cesium.Cartesian3.fromDegreesArray(seg.coords),
            width: 6,
            material: Cesium.Color.fromCssColorString(seg.severityColor),
            clampToGround: true,
          },
          properties: seg,
        }),
      );
    }
  }

  return {
    init() {},
    enable() { enabled = true; return update(); },
    disable() {
      enabled = false;
      entities.forEach((e) => viewer.entities.remove(e));
      entities.length = 0;
    },
    update,
    destroy() { this.disable(); },
    getStats() { return { count: entities.length, provider: 'Veabird + CONANP' }; },
  };
}
